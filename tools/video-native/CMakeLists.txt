cmake_minimum_required(VERSION 3.23)
project(VideoNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(VideoNative WIN32 src/main.cpp)
target_compile_definitions(VideoNative PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(VideoNative PRIVATE mfplat mfreadwrite mfuuid mf ole32 shlwapi bcrypt comctl32)
if(MSVC)
 target_compile_options(VideoNative PRIVATE /W3 /EHsc)
endif()
